    yield;
    declare PanelClicked = False;
    foreach (Evt in PendingEvents) {
      if (Evt.Type != CMapEditorPluginEvent::Type::LayerCustomEvent || Evt.CustomEventData.count == 0) {
        continue;
      }
      if (Evt.CustomEventType == "AtlasPanelHover") {
        G_PanelHovered = Evt.CustomEventData[0] == "1";
        continue;
      }
      if (Evt.CustomEventType != "AtlasPanel") {
        continue;
      }
      PanelClicked = True;
      declare Action = Evt.CustomEventData[0];
      if (Action == "AtlasGround") {
        G_SelectedGroundGroup = "BayDocks";
        Atlas::SetGroundSelectionGroup(G_SelectedGroundGroup);
        Atlas::SetSelectionMode(Atlas::C_SelectionMode_Ground2D);
      } else if (Action == "AtlasEsplanade") {
        G_SelectedGroundGroup = "BayEsplanade";
        Atlas::SetGroundSelectionGroup(G_SelectedGroundGroup);
        Atlas::SetSelectionMode(Atlas::C_SelectionMode_Ground2D);
      } else {
        if (Action == "AtlasPlane") {
          Atlas::SetSelectionMode(Atlas::C_SelectionMode_Plane2D);
        } else if (Action == "AtlasBox") {
          Atlas::SetSelectionMode(Atlas::C_SelectionMode_Box3D);
        } else {
          if (Action == "AtlasLine") {
            Atlas::SetSelectionMode(Atlas::C_SelectionMode_Line1D);
          } else if (Action == "AtlasRemoveWater") {
            Atlas::SetSelectionMode(Atlas::C_SelectionMode_RemoveWater);
          } else {
            if (Action == "AtlasRestoreWater") {
              Atlas::SetSelectionMode(Atlas::C_SelectionMode_RestoreWater);
            } else if (Action == "AtlasRemoveDocks" || Action == "AtlasRemoveEsplanade") {
              if (Action == "AtlasRemoveDocks") {
                G_SelectedGroundGroup = "BayDocks";
              } else {
                G_SelectedGroundGroup = "BayEsplanade";
              }
              Atlas::SetGroundSelectionGroup(G_SelectedGroundGroup);
              Atlas::SetSelectionMode(Atlas::C_SelectionMode_RemoveItemGroup);
            } else {
              if (Action == "AtlasNone") {
                Atlas::SetSelectionMode(Atlas::C_SelectionMode_None);
              } else if (Action == "AtlasConnections") {
                if (Atlas::GetFreeformMode() == Atlas::C_FreeformPlacementMode_ConnectExisting) {
                  Atlas::SetFreeformPlacementMode(Atlas::C_FreeformPlacementMode_SelectionOnly);
                } else {
                  Atlas::SetFreeformPlacementMode(Atlas::C_FreeformPlacementMode_ConnectExisting);
                }
              } else {
                if (Action == "AtlasUndo") {
                  if (!Atlas::GetCanUndoAtlasEdit()) {
                    log("No item edit to undo.");
                  } else if (!Atlas::UndoAtlasEdit()) {
                    log("Atlas undo failed.");
                  }
                } else if (Action == "AtlasRedo") {
                  if (!Atlas::GetCanRedoAtlasEdit()) {
                    log("No item edit to redo.");
                  } else if (!Atlas::RedoAtlasEdit()) {
                    log("Atlas redo failed.");
                  }
                }
              }
            }
          }
        }
      }
    }
    Atlas::SetSelectionInputEnabled(!G_PanelHovered && !PanelClicked);
    Atlas::Update();
    foreach (Change in Atlas::GetSelectionConfirmed()) {
      if (Atlas::GetMode() == Atlas::C_SelectionMode_Ground2D) {
        if (!Atlas::PlaceFreeform1x1(Change.Coords, G_SelectedGroundGroup)) {
          log(G_SelectedGroundGroup ^ " placement failed.");
        }
      } else if (Atlas::GetMode() == Atlas::C_SelectionMode_RemoveItemGroup) {
        if (!Atlas::RemoveFreeform1x1(Change.Coords, G_SelectedGroundGroup)) {
          log(G_SelectedGroundGroup ^ " removal failed.");
        }
      } else {
        if (Atlas::GetMode() != Atlas::C_SelectionMode_RemoveWater && Atlas::GetMode() != Atlas::C_SelectionMode_RestoreWater) {
          Atlas::SetCurrentSelection(Change.Coords, True);
        }
      }
    }
    foreach (Removal in Atlas::GetItemRemovals()) {
      log("Item removed at " ^ Removal.Position);
    }
  }
