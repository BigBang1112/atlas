#RequireContext CMapEditorPlugin

#Include "Libs/AtlasEngine.Script.txt" as Atlas

#Struct Atlas::SelectionChange as SelectionChange
#Struct Atlas::ItemRemoval as ItemRemoval

#Const C_Atlas_Libs_AtlasEngine_SelectionMode_None 0
#Const C_Atlas_Libs_AtlasEngine_SelectionMode_Box3D 1
#Const C_Atlas_Libs_AtlasEngine_SelectionMode_BoxToGround 2
#Const C_Atlas_Libs_AtlasEngine_SelectionMode_Plane2D 3
#Const C_Atlas_Libs_AtlasEngine_SelectionMode_Ground2D 4
#Const C_Atlas_Libs_AtlasEngine_SelectionMode_Line1D 5
#Const C_Atlas_Libs_AtlasEngine_SelectionMode_Tower 6
#Const C_Atlas_Libs_AtlasEngine_SelectionMode_RemoveWater 7
#Const C_Atlas_Libs_AtlasEngine_SelectionMode_RestoreWater 8

Void SelectGround() {
    Atlas::SetSelectionMode(C_Atlas_Libs_AtlasEngine_SelectionMode_Ground2D);
}

Void RemoveWater() {
    Atlas::SetSelectionMode(C_Atlas_Libs_AtlasEngine_SelectionMode_RemoveWater);
}

Void RestoreWater() {
    Atlas::SetSelectionMode(C_Atlas_Libs_AtlasEngine_SelectionMode_RestoreWater);
}

main() {
    Atlas::SetRemoveWaterBlockMapping(["Grass" => "LagoonGrassVoid", "Beach" => "LagoonBeachVoid"]);
    Atlas::SetRestoreWaterBlockMapping(["LagoonGrassVoid", "LagoonBeachVoid"], "LagoonVoid");
    Atlas::Initialize();
    Atlas::SetSelectionMode(C_Atlas_Libs_AtlasEngine_SelectionMode_Ground2D);
    log("Atlas example ready: drag on the ground to select cells.");
    while (True) {
        yield;
        Atlas::Update();
        foreach (Change in Atlas::GetSelectionChanged()) {
            log("Selection preview: " ^ Change.Coords.count ^ " cells");
        }
        foreach (Change in Atlas::GetSelectionConfirmed()) {
            log("Selection confirmed: " ^ Change.Coords.count ^ " cells");
        }
        foreach (Removal in Atlas::GetItemRemovals()) {
            log("Item removed at " ^ Removal.Position);
        }
    }
}
