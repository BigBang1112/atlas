Boolean Private_SameCoord(Int3 _A, Int3 _B) {
  return _A.X == _B.X && _A.Y == _B.Y && _A.Z == _B.Z;
}

Boolean Private_SamePosition(Vec3 _A, Vec3 _B) {
  return _A.X == _B.X && _A.Y == _B.Y && _A.Z == _B.Z;
}

Void Private_ResetSelectionChangeTracking() {
  G_LastSelectionChange.clear();
  G_HasLastSelectionChange = False;
}

Boolean Private_SelectionChangedSinceLastEvent(Int3[] _Coords) {
  if (G_HasLastSelectionChange && G_LastSelectionChange.count == _Coords.count) {
    declare Matches = True;
    for (I, 0, _Coords.count - 1) {
      if (!Private_SameCoord(G_LastSelectionChange[I], _Coords[I])) {
        Matches = False;
        break;
      }
    }
    if (Matches) {
      return False;
    }
  }
  G_LastSelectionChange.clear();
  foreach (Coord in _Coords) {
    G_LastSelectionChange.add(Coord);
  }
  G_HasLastSelectionChange = True;
  return True;
}

Boolean Private_SameItemBlock(ItemBlock _A, ItemBlock _B) {
  return _A.MacroblockName == _B.MacroblockName && Private_SameCoord(_A.MacroblockCoord, _B.MacroblockCoord) && _A.MacroblockDir == _B.MacroblockDir && Private_SamePosition(_A.ItemPosition, _B.ItemPosition) && _A.Ground == _B.Ground && _A.Family == _B.Family && _A.PieceIndex == _B.PieceIndex && _A.Width == _B.Width && _A.Depth == _B.Depth;
}

Integer DirectionToIndex(CMapEditorPlugin::CardinalDirections _Direction) {
  if (_Direction == CMapEditorPlugin::CardinalDirections::East) {
    return 1;
  }
  if (_Direction == CMapEditorPlugin::CardinalDirections::South) {
    return 2;
  }
  if (_Direction == CMapEditorPlugin::CardinalDirections::West) {
    return 3;
  }
  return 0;
}

Boolean Private_SamePlacement(ItemBlock _Block, Placement _Placement) {
  return Private_SameCoord(_Block.MacroblockCoord, _Placement.Coord) && _Block.MacroblockName == _Placement.MacroblockName && _Block.MacroblockDir == DirectionToIndex(_Placement.Direction) && _Block.Ground == _Placement.Ground && _Block.Family == _Placement.Family && _Block.PieceIndex == _Placement.PieceIndex && _Block.Width == _Placement.Width && _Block.Depth == _Placement.Depth;
}

Integer Private_IndexOfItemBlock(ItemBlock[] _Blocks, ItemBlock _Target) {
  for (Index, 0, _Blocks.count - 1) {
    if (Private_SameItemBlock(_Blocks[Index], _Target)) {
      return Index;
    }
  }
  return -1;
}

ItemBlock[] Private_CopyItemBlocks(ItemBlock[] _Blocks) {
  declare ItemBlock[] Copy;
  foreach (Block in _Blocks) {
    Copy.add(Block);
  }
  return Copy;
}

Boolean Private_ContainsItemBlocks(ItemBlock[] _Blocks, ItemBlock[] _Expected) {
  declare Unmatched = Private_CopyItemBlocks(_Blocks);
  foreach (Block in _Expected) {
    declare Index = Private_IndexOfItemBlock(Unmatched, Block);
    if (Index < 0) {
      return False;
    }
    Unmatched.removekey(Index);
  }
  return True;
}

Boolean Private_MatchesItemBlockChanges(ItemBlock[] _Blocks, ItemBlock[] _ExpectedPresent, ItemBlock[] _ExpectedAbsent) {
  if (!Private_ContainsItemBlocks(_Blocks, _ExpectedPresent)) {
    return False;
  }
  foreach (Block in _ExpectedAbsent) {
    if (Private_IndexOfItemBlock(_Blocks, Block) >= 0) {
      return False;
    }
  }
  return True;
}

Void Private_PushUndoEdit(AtlasEdit _Edit) {
  G_UndoEdits.add(_Edit);
  if (G_UndoEdits.count > 64) {
    G_UndoEdits.removekey(0);
  }
  G_RedoEdits.clear();
}

Void Private_PushItemEdit(ItemBlock[] _Removed, ItemBlock[] _Added, NoItemReplacement[] _RemovedNoItemBlocks) {
  if (G_ReplayingHistory || (_Removed.count == 0 && _Added.count == 0)) {
    return;
  }
  declare NoItemReplacement[] Placeholders;
  foreach (Entry in _RemovedNoItemBlocks) {
    Placeholders.add(Entry);
  }
  Private_PushUndoEdit(AtlasEdit { AddedBlocks = Private_CopyItemBlocks(_Added), RemovedBlocks = Private_CopyItemBlocks(_Removed), RemovedNoItemBlocks = Placeholders, NoItemBlockName = G_NoItemBlockName });
}

Void ClearAtlasEditHistory() {
  G_UndoEdits.clear();
  G_RedoEdits.clear();
}

CMapEditorPlugin::CardinalDirections DirectionFromIndex(Integer _Index) {
  if (_Index == 1) {
    return CMapEditorPlugin::CardinalDirections::East;
  }
  if (_Index == 2) {
    return CMapEditorPlugin::CardinalDirections::South;
  }
  if (_Index == 3) {
    return CMapEditorPlugin::CardinalDirections::West;
  }
  return CMapEditorPlugin::CardinalDirections::North;
}

Void SetItemBlockList(ItemBlock[] _Blocks) {
  declare ItemBlock[] Copy;
  foreach (Block in _Blocks) {
    Copy.add(Block);
  }
  declare metadata ItemBlock[] Atlas_ItemBlocks as Stored for Map;
  Stored/* not Null here */.clear();
  foreach (AddRangeItem in Copy) {
    Stored.add(AddRangeItem);
  }
  G_GroundItemHeightsValid = False;
}

Void Private_SnapshotItems() {
  if (G_ReplayingHistory) {
    G_DeferredItemSnapshotUpdates = 2;
    return;
  }
  G_PreviousItems.clear();
  foreach (Item in Items) {
    if (Item != Null) {
      G_PreviousItems.add(Item.Position);
    }
  }
}

Void Private_RestoreRemoved(ItemBlock[] _Removed) {
  declare metadata ItemBlock[] Atlas_ItemBlocks as StoredBlocks for Map;
  foreach (Block in _Removed) {
    declare Model = GetMacroblockModelFromFilePath(Block.MacroblockName);
    if (Model != Null && PlaceMacroblock_NoDestruction(Model, Block.MacroblockCoord, DirectionFromIndex(Block.MacroblockDir))) {
      StoredBlocks/* not Null here */.add(Block);
    } else {
      G_LastRollbackFailed = True;
    }
  }
  Private_SnapshotItems();
}

Boolean Private_WithinMap(Int3 _Coord) {
  return _Coord.X >= 0 && _Coord.Z >= 0 && _Coord.Y >= 0 && _Coord.X < Map.Size.X && _Coord.Z < Map.Size.Z && _Coord.Y < Map.Size.Y;
}

Void Private_RestoreNoItemBlocks(NoItemReplacement[] _Removed) {
  if (_Removed.count == 0) {
    return;
  }
  declare MacroblockModel = GetMacroblockModelFromFilePath(G_NoItemBlockName);
  if (MacroblockModel != Null) {
    foreach (Entry in _Removed) {
      if (!PlaceMacroblock_NoDestruction(MacroblockModel, Entry.Coord, Entry.Direction)) {
        log("Could not restore no-item macroblock '" ^ G_NoItemBlockName ^ "' at " ^ Entry.Coord ^ ".");
        G_LastRollbackFailed = True;
      }
    }
    return;
  }
  declare BlockModel = GetBlockModelFromName(G_NoItemBlockName);
  foreach (Entry in _Removed) {
    if (BlockModel == Null || !PlaceBlock(BlockModel, Entry.Coord, Entry.Direction)) {
      log("Could not restore no-item block '" ^ G_NoItemBlockName ^ "' at " ^ Entry.Coord ^ ".");
      G_LastRollbackFailed = True;
    }
  }
}

Void Private_RollbackPlacementPlan(ItemBlock[] _Removed, ItemBlock[] _Placed, NoItemReplacement[] _RemovedNoItemBlocks) {
  declare ItemBlock[] FailedToRemove;
  declare Index = _Placed.count - 1;
  while (Index >= 0) {
    declare Block = _Placed[Index];
    declare Model = GetMacroblockModelFromFilePath(Block.MacroblockName);
    if (Model == Null || !RemoveMacroblock(Model, Block.MacroblockCoord, DirectionFromIndex(Block.MacroblockDir))) {
      FailedToRemove.add(Block);
      G_LastRollbackFailed = True;
    }
    Index -= 1;
  }
  declare ItemBlock[] FailedToRestore;
  foreach (Block in _Removed) {
    declare Model = GetMacroblockModelFromFilePath(Block.MacroblockName);
    if (Model == Null || !PlaceMacroblock_NoDestruction(Model, Block.MacroblockCoord, DirectionFromIndex(Block.MacroblockDir))) {
      FailedToRestore.add(Block);
      G_LastRollbackFailed = True;
    }
  }
  Private_RestoreNoItemBlocks(_RemovedNoItemBlocks);
  if (G_LastRollbackFailed) {
    declare ItemBlock[] Updated;
    foreach (Block in GetItemBlockList()) {
      if (Private_IndexOfItemBlock(FailedToRestore, Block) < 0) {
        Updated.add(Block);
      }
    }
    foreach (Block in FailedToRemove) {
      Updated.add(Block);
    }
    SetItemBlockList(Updated);
  }
  Private_SnapshotItems();
}

NoItemRemoval Private_RemoveNoItemBlocks(Int3 _Coord, Integer _Width, Integer _Depth, Boolean _Ground) {
  declare NoItemReplacement[] Removed;
  if (!_Ground || G_NoItemBlockName == "") {
    return NoItemRemoval { Success = True, Removed = Removed };
  }
  declare MacroblockModel = GetMacroblockModelFromFilePath(G_NoItemBlockName);
  if (MacroblockModel == Null && GetBlockModelFromName(G_NoItemBlockName) == Null) {
    log("No item block '" ^ G_NoItemBlockName ^ "' was not found.");
    return NoItemRemoval { Success = False, Removed = Removed };
  }
  for (X, _Coord.X, _Coord.X + _Width - 1) {
    for (Z, _Coord.Z, _Coord.Z + _Depth - 1) {
      declare Cell = <X, CollectionGroundY, Z>;
      if (MacroblockModel != Null) {
        declare Heights = [CollectionGroundY];
        if (_Coord.Y != CollectionGroundY) {
          Heights.add(_Coord.Y);
        }
        declare RemovedAtCell = False;
        foreach (Y in Heights) {
          if (RemovedAtCell) {
            break;
          }
          declare Candidate = <X, Y, Z>;
          for (Direction, 0, 4 - 1) {
            if (!RemoveMacroblock(MacroblockModel, Candidate, DirectionFromIndex(Direction))) {
              continue;
            }
            Removed.add(NoItemReplacement { Coord = Candidate, Direction = DirectionFromIndex(Direction) });
            RemovedAtCell = True;
            break;
          }
        }
        continue;
      }
      declare Block = GetBlock(Cell);
      if (Block == Null || Block.BlockModel.Name != G_NoItemBlockName) {
        continue;
      }
      if (!RemoveBlock(Cell)) {
        return NoItemRemoval { Success = False, Removed = Removed };
      }
      declare BlockDirection = CMapEditorPlugin::CardinalDirections::North;
      if (Block.Direction == CBlock::CardinalDirections::East) {
        BlockDirection = CMapEditorPlugin::CardinalDirections::East;
      } else if (Block.Direction == CBlock::CardinalDirections::South) {
        BlockDirection = CMapEditorPlugin::CardinalDirections::South;
      } else {
        if (Block.Direction == CBlock::CardinalDirections::West) {
          BlockDirection = CMapEditorPlugin::CardinalDirections::West;
        }
      }
      Removed.add(NoItemReplacement { Coord = Cell, Direction = BlockDirection });
    }
  }
  return NoItemRemoval { Success = True, Removed = Removed };
}

Void Private_UpdateGroundItemHeights(ItemBlock[] _Removed, ItemBlock[] _Added) {
  if (!G_GroundItemHeightsValid) {
    return;
  }
  foreach (Block in _Removed) {
    if (Block.Ground) {
      G_GroundItemHeightsValid = False;
      return;
    }
  }
  foreach (Block in _Added) {
    if (!Block.Ground) {
      continue;
    }
    for (CellX, Block.MacroblockCoord.X, Block.MacroblockCoord.X + MathLib::Max(1, Block.Width) - 1) {
      for (CellZ, Block.MacroblockCoord.Z, Block.MacroblockCoord.Z + MathLib::Max(1, Block.Depth) - 1) {
        declare Cell = <CellX, 0, CellZ>;
        if (!G_GroundItemHeights.existskey(Cell) || G_GroundItemHeights[Cell] < Block.MacroblockCoord.Y) {
          G_GroundItemHeights[Cell] = Block.MacroblockCoord.Y;
        }
      }
    }
  }
}

Boolean ExecutePlacementPlan(Placement[] _Plan) {
  G_LastRollbackFailed = False;
  if (_Plan.count == 0) {
    return False;
  }
  declare metadata ItemBlock[] Atlas_ItemBlocks as StoredBlocks for Map;
  declare Existing = StoredBlocks/* not Null here */;
  declare Boolean[Int3] PlannedCells;
  declare MinX = _Plan[0].Coord.X;
  declare MaxX = MinX;
  declare MinY = _Plan[0].Coord.Y;
  declare MaxY = MinY;
  declare MinZ = _Plan[0].Coord.Z;
  declare MaxZ = MinZ;
  foreach (Entry in _Plan) {
    if (!Private_WithinMap(Entry.Coord) || GetMacroblockModelFromFilePath(Entry.MacroblockName) == Null) {
      return False;
    }
    declare Width = MathLib::Max(1, Entry.Width);
    declare Depth = MathLib::Max(1, Entry.Depth);
    MinX = MathLib::Min(MinX, Entry.Coord.X);
    MaxX = MathLib::Max(MaxX, Entry.Coord.X + Width - 1);
    MinY = MathLib::Min(MinY, Entry.Coord.Y);
    MaxY = MathLib::Max(MaxY, Entry.Coord.Y);
    MinZ = MathLib::Min(MinZ, Entry.Coord.Z);
    MaxZ = MathLib::Max(MaxZ, Entry.Coord.Z + Depth - 1);
    for (X, Entry.Coord.X, Entry.Coord.X + Width - 1) {
      for (Z, Entry.Coord.Z, Entry.Coord.Z + Depth - 1) {
        declare Cell = <X, Entry.Coord.Y, Z>;
        if (!Private_WithinMap(Cell) || PlannedCells.existskey(Cell)) {
          return False;
        }
        PlannedCells[Cell] = True;
      }
    }
  }
  declare Integer[][Int3] ExistingByCell;
  for (Index, 0, Existing.count - 1) {
    declare Block = Existing[Index];
    if (Block.MacroblockCoord.Y < MinY || Block.MacroblockCoord.Y > MaxY || Block.MacroblockCoord.X > MaxX || Block.MacroblockCoord.X + MathLib::Max(1, Block.Width) <= MinX || Block.MacroblockCoord.Z > MaxZ || Block.MacroblockCoord.Z + MathLib::Max(1, Block.Depth) <= MinZ) {
      continue;
    }
    for (X, Block.MacroblockCoord.X, Block.MacroblockCoord.X + MathLib::Max(1, Block.Width) - 1) {
      for (Z, Block.MacroblockCoord.Z, Block.MacroblockCoord.Z + MathLib::Max(1, Block.Depth) - 1) {
        declare Cell = <X, Block.MacroblockCoord.Y, Z>;
        if (!PlannedCells.existskey(Cell)) {
          continue;
        }
        if (!ExistingByCell.existskey(Cell)) {
          ExistingByCell[Cell] = [];
        }
        ExistingByCell[Cell].add(Index);
      }
    }
  }
  declare Boolean[Integer] RemovedIndices;
  foreach (Entry in _Plan) {
    declare Width = MathLib::Max(1, Entry.Width);
    declare Depth = MathLib::Max(1, Entry.Depth);
    for (X, Entry.Coord.X, Entry.Coord.X + Width - 1) {
      for (Z, Entry.Coord.Z, Entry.Coord.Z + Depth - 1) {
        declare Cell = <X, Entry.Coord.Y, Z>;
        if (!ExistingByCell.existskey(Cell)) {
          continue;
        }
        foreach (Index in ExistingByCell[Cell]) {
          declare Block = Existing[Index];
          if (Block.Family != Entry.Family) {
            return False;
          }
          if (Private_SamePlacement(Block, Entry)) {
            continue;
          }
          RemovedIndices[Index] = True;
        }
      }
    }
  }
  declare ItemBlock[] Removed;
  for (Index, 0, Existing.count - 1) {
    if (!RemovedIndices.existskey(Index)) {
      continue;
    }
    declare Block = Existing[Index];
    declare Model = GetMacroblockModelFromFilePath(Block.MacroblockName);
    if (Model == Null || !RemoveMacroblock(Model, Block.MacroblockCoord, DirectionFromIndex(Block.MacroblockDir))) {
      Private_RollbackPlacementPlan(Removed, [], []);
      return False;
    }
    Removed.add(Block);
  }
  declare ItemBlock[] Placed;
  declare NoItemReplacement[] RemovedNoItemBlocks;
  foreach (Entry in _Plan) {
    declare AlreadyThere = False;
    if (ExistingByCell.existskey(Entry.Coord)) {
      foreach (Index in ExistingByCell[Entry.Coord]) {
        declare Block = Existing[Index];
        if (!RemovedIndices.existskey(Index) && Private_SamePlacement(Block, Entry)) {
          AlreadyThere = True;
        }
      }
    }
    if (AlreadyThere) {
      continue;
    }
    declare Model = GetMacroblockModelFromFilePath(Entry.MacroblockName);
    declare RemovalResult = Private_RemoveNoItemBlocks(Entry.Coord, MathLib::Max(1, Entry.Width), MathLib::Max(1, Entry.Depth), Entry.Ground);
    foreach (Cell in RemovalResult.Removed) {
      RemovedNoItemBlocks.add(Cell);
    }
    if (!RemovalResult.Success) {
      Private_RollbackPlacementPlan(Removed, Placed, RemovedNoItemBlocks);
      return False;
    }
    if (Model == Null || !CanPlaceMacroblock_NoDestruction(Model, Entry.Coord, Entry.Direction)) {
      Private_RollbackPlacementPlan(Removed, Placed, RemovedNoItemBlocks);
      return False;
    }
    declare PreviousCount = Items.count;
    if (!PlaceMacroblock_NoDestruction(Model, Entry.Coord, Entry.Direction)) {
      Private_RollbackPlacementPlan(Removed, Placed, RemovedNoItemBlocks);
      return False;
    }
    declare Vec3 Position;
    if (!G_ReplayingHistory && Items.count > PreviousCount) {
      Position = Items[PreviousCount].Position;
    } else {
      Position = GetVec3FromCoord(Entry.Coord);
    }
    Placed.add(ItemBlock { MacroblockName = Entry.MacroblockName, MacroblockCoord = Entry.Coord, MacroblockDir = DirectionToIndex(Entry.Direction), ItemPosition = Position, Ground = Entry.Ground, Family = Entry.Family, PieceIndex = Entry.PieceIndex, Width = Entry.Width, Depth = Entry.Depth });
  }
  if (Removed.count > 0 || Placed.count > 0) {
    declare Index = StoredBlocks/* not Null here */.count - 1;
    while (Index >= 0) {
      if (RemovedIndices.existskey(Index)) {
        StoredBlocks.removekey(Index);
      }
      Index -= 1;
    }
    foreach (Block in Placed) {
      StoredBlocks/* not Null here */.add(Block);
    }
    Private_UpdateGroundItemHeights(Removed, Placed);
    Private_SnapshotItems();
    Private_PushItemEdit(Removed, Placed, RemovedNoItemBlocks);
  }
  return True;
}

Void SetRemovedWater(Int3[] _Coords) {
  if (!G_ReplayingHistory) {
    ClearAtlasEditHistory();
  }
  declare Int3[] Copy;
  foreach (Coord in _Coords) {
    Copy.add(Coord);
  }
  declare metadata Int3[] Atlas_RemovedWater as Stored for Map;
  Stored/* not Null here */.clear();
  foreach (Coord in Copy) {
    Stored.add(Coord);
  }
  G_GroundItemHeightsValid = False;
}

Boolean Private_ApplyResolvedChangesCore(ItemBlock[] _ToRemove, Placement[] _ToPlace, Int3[] _RemovedWater, Boolean _ApplyWater) {
  G_LastRollbackFailed = False;
  declare ItemBlock[] Removed;
  declare metadata ItemBlock[] Atlas_ItemBlocks as StoredBlocks for Map;
  declare Remaining = Private_CopyItemBlocks(StoredBlocks/* not Null here */);
  foreach (Block in _ToRemove) {
    declare Index = Private_IndexOfItemBlock(Remaining, Block);
    if (Index < 0) {
      if (Removed.count > 0) {
        SetItemBlockList(Remaining);
      }
      Private_RestoreRemoved(Removed);
      return False;
    }
    declare Model = GetMacroblockModelFromFilePath(Block.MacroblockName);
    if (Model == Null || !RemoveMacroblock(Model, Block.MacroblockCoord, DirectionFromIndex(Block.MacroblockDir))) {
      if (Removed.count > 0) {
        SetItemBlockList(Remaining);
      }
      Private_RestoreRemoved(Removed);
      return False;
    }
    Remaining.removekey(Index);
    Removed.add(Block);
  }
  if (Removed.count > 0) {
    SetItemBlockList(Remaining);
  }
  if (_ToPlace.count > 0 && !ExecutePlacementPlan(_ToPlace)) {
    Private_RestoreRemoved(Removed);
    return False;
  }
  if (_ApplyWater) {
    SetRemovedWater(_RemovedWater);
  }
  Private_SnapshotItems();
  return True;
}

Int3[] GetRemovedWater() {
  declare metadata Int3[] Atlas_RemovedWater as Stored for Map;
  return Stored/* not Null here */;
}

Boolean Private_ApplyItemChanges(ItemBlock[] _ToRemove, ItemBlock[] _ToPlaceBlocks) {
  declare Placement[] ToPlace;
  foreach (Block in _ToPlaceBlocks) {
    ToPlace.add(Placement { Coord = Block.MacroblockCoord, MacroblockName = Block.MacroblockName, Direction = DirectionFromIndex(Block.MacroblockDir), Ground = Block.Ground, Family = Block.Family, PieceIndex = Block.PieceIndex, Width = Block.Width, Depth = Block.Depth });
  }
  if (!Private_ApplyResolvedChangesCore(_ToRemove, ToPlace, GetRemovedWater(), False)) {
    return False;
  }
  Private_SnapshotItems();
  return True;
}

Boolean GetCanUndoAtlasEdit() {
  return G_UndoEdits.count > 0;
}

Boolean UndoAtlasEdit() {
  if (!GetCanUndoAtlasEdit()) {
    return False;
  }
  declare Edit = G_UndoEdits[G_UndoEdits.count - 1];
  if (!Private_MatchesItemBlockChanges(GetItemBlockList(), Edit.AddedBlocks, Edit.RemovedBlocks)) {
    ClearAtlasEditHistory();
    return False;
  }
  declare ConfiguredNoItemBlockName = G_NoItemBlockName;
  G_NoItemBlockName = Edit.NoItemBlockName;
  G_ReplayingHistory = True;
  declare Succeeded = Private_ApplyItemChanges(Edit.AddedBlocks, Edit.RemovedBlocks);
  if (Succeeded) {
    G_LastRollbackFailed = False;
    log("Atlas undo: restoring " ^ Edit.RemovedNoItemBlocks.count ^ " no-item block(s).");
    Private_RestoreNoItemBlocks(Edit.RemovedNoItemBlocks);
    Succeeded = !G_LastRollbackFailed;
    if (!Succeeded) {
      Private_ApplyItemChanges(Edit.RemovedBlocks, Edit.AddedBlocks);
    } else {
      Private_SnapshotItems();
    }
  }
  G_ReplayingHistory = False;
  G_NoItemBlockName = ConfiguredNoItemBlockName;
  if (!Succeeded) {
    if (G_LastRollbackFailed) {
      ClearAtlasEditHistory();
    }
    return False;
  }
  G_UndoEdits.removekey(G_UndoEdits.count - 1);
  G_RedoEdits.add(Edit);
  return True;
}

Boolean GetCanRedoAtlasEdit() {
  return G_RedoEdits.count > 0;
}

Boolean RedoAtlasEdit() {
  if (!GetCanRedoAtlasEdit()) {
    return False;
  }
  declare Edit = G_RedoEdits[G_RedoEdits.count - 1];
  if (!Private_MatchesItemBlockChanges(GetItemBlockList(), Edit.RemovedBlocks, Edit.AddedBlocks)) {
    ClearAtlasEditHistory();
    return False;
  }
  declare ConfiguredNoItemBlockName = G_NoItemBlockName;
  G_NoItemBlockName = Edit.NoItemBlockName;
  G_ReplayingHistory = True;
  declare Succeeded = Private_ApplyItemChanges(Edit.RemovedBlocks, Edit.AddedBlocks);
  G_ReplayingHistory = False;
  G_NoItemBlockName = ConfiguredNoItemBlockName;
  if (!Succeeded) {
    if (G_LastRollbackFailed) {
      ClearAtlasEditHistory();
    }
    return False;
  }
  G_RedoEdits.removekey(G_RedoEdits.count - 1);
  G_UndoEdits.add(Edit);
  return True;
}

Integer GetFakeGroundHeight(Integer _X, Integer _Z) {
  if (!G_GroundItemHeightsValid) {
    G_GroundItemHeights.clear();
    G_RemovedWaterGroundHeights.clear();
    foreach (Coord in GetRemovedWater()) {
      declare Cell = <Coord.X, 0, Coord.Z>;
      if (!G_RemovedWaterGroundHeights.existskey(Cell) || Coord.Y < G_RemovedWaterGroundHeights[Cell]) {
        G_RemovedWaterGroundHeights[Cell] = Coord.Y;
      }
    }
    foreach (Block in GetItemBlockList()) {
      if (!Block.Ground) {
        continue;
      }
      for (CellX, Block.MacroblockCoord.X, Block.MacroblockCoord.X + MathLib::Max(1, Block.Width) - 1) {
        for (CellZ, Block.MacroblockCoord.Z, Block.MacroblockCoord.Z + MathLib::Max(1, Block.Depth) - 1) {
          declare Cell = <CellX, 0, CellZ>;
          if (!G_GroundItemHeights.existskey(Cell) || G_GroundItemHeights[Cell] < Block.MacroblockCoord.Y) {
            G_GroundItemHeights[Cell] = Block.MacroblockCoord.Y;
          }
        }
      }
    }
    G_GroundItemHeightsValid = True;
  }
  declare Key = <_X, 0, _Z>;
  declare Integer Height;
  if (G_RemovedWaterGroundHeights.existskey(Key)) {
    Height = G_RemovedWaterGroundHeights[Key];
  } else {
    Height = GetGroundHeight(_X, _Z);
  }
  if (G_GroundItemHeights.existskey(Key) && G_GroundItemHeights[Key] > Height) {
    Height = G_GroundItemHeights[Key];
  }
  return Height;
}

ItemBlock[] Private_RemoveItemBlocksAtPosition(Vec3 _Position) {
  declare metadata ItemBlock[] Atlas_ItemBlocks as StoredBlocks for Map;
  declare ItemBlock[] Remaining;
  declare ItemBlock[] Removed;
  foreach (Block in StoredBlocks/* not Null here */) {
    if (Private_SamePosition(Block.ItemPosition, _Position)) {
      declare Model = GetMacroblockModelFromFilePath(Block.MacroblockName);
      if (Model != Null && RemoveMacroblock(Model, Block.MacroblockCoord, DirectionFromIndex(Block.MacroblockDir))) {
        Removed.add(Block);
        continue;
      }
    }
    Remaining.add(Block);
  }
  if (Removed.count > 0) {
    SetItemBlockList(Remaining);
  }
  return Removed;
}

Void SyncManuallyRemovedItems() {
  if (G_DeferredItemSnapshotUpdates > 0) {
    G_DeferredItemSnapshotUpdates -= 1;
    if (G_DeferredItemSnapshotUpdates == 0) {
      Private_SnapshotItems();
    }
    return;
  }
  if (Items.count > G_PreviousItems.count) {
    Private_SnapshotItems();
    return;
  }
  if (Items.count == G_PreviousItems.count) {
    return;
  }
  declare Integer[Vec3] Counts;
  foreach (Item in Items) {
    if (Item == Null) {
      continue;
    }
    if (!Counts.existskey(Item.Position)) {
      Counts[Item.Position] = 0;
    }
    Counts[Item.Position] += 1;
  }
  declare Vec3[] RemovedPositions;
  declare I = G_PreviousItems.count - 1;
  while (I >= 0) {
    declare Position = G_PreviousItems[I];
    if (!Counts.existskey(Position) || Counts[Position] == 0) {
      RemovedPositions.add(Position);
      G_PreviousItems.removekey(I);
    } else {
      Counts[Position] -= 1;
    }
    I -= 1;
  }
  foreach (Position in RemovedPositions) {
    declare RemovedBlocks = Private_RemoveItemBlocksAtPosition(Position);
    G_ItemRemovals.add(ItemRemoval { Position = Position, RemovedBlocks = RemovedBlocks });
  }
  Private_SnapshotItems();
}

Int3[] BuildSelection(Int3 _Start, Int3 _End, Integer _SelectionMode) {
  declare Int3[] Result;
  declare MinX = MathLib::Min(_Start.X, _End.X);
  declare MaxX = MathLib::Max(_Start.X, _End.X);
  declare MinZ = MathLib::Min(_Start.Z, _End.Z);
  declare MaxZ = MathLib::Max(_Start.Z, _End.Z);
  if (_SelectionMode == C_SelectionMode_Line1D) {
    declare Dx = MathLib::Abs(_End.X - _Start.X);
    declare Dy = MathLib::Abs(_End.Y - _Start.Y);
    declare Dz = MathLib::Abs(_End.Z - _Start.Z);
    declare Count = MathLib::Max(Dx, MathLib::Max(Dy, Dz));
    for (I, 0, Count) {
      if (Dx >= Dy && Dx >= Dz) {
        declare Step = I;
        if (_End.X < _Start.X) {
          Step = -I;
        }
        Result.add(<_Start.X + Step, _Start.Y, _Start.Z>);
      } else if (Dy >= Dz) {
        declare Step = I;
        if (_End.Y < _Start.Y) {
          Step = -I;
        }
        Result.add(<_Start.X, _Start.Y + Step, _Start.Z>);
      } else {
        declare Step = I;
        if (_End.Z < _Start.Z) {
          Step = -I;
        }
        Result.add(<_Start.X, _Start.Y, _Start.Z + Step>);
      }
    }
    return Result;
  }
  declare GroundForBox = 0;
  if (_SelectionMode == C_SelectionMode_BoxToGround) {
    GroundForBox = GetFakeGroundHeight(MinX, MinZ);
  }
  for (X, MinX, MaxX) {
    for (Z, MinZ, MaxZ) {
      if (_SelectionMode == C_SelectionMode_Ground2D || _SelectionMode == C_SelectionMode_RemoveWater || _SelectionMode == C_SelectionMode_RestoreWater) {
        declare Integer Y;
        if (_SelectionMode == C_SelectionMode_Ground2D) {
          Y = GetFakeGroundHeight(X, Z);
        } else {
          Y = CollectionGroundY;
        }
        Result.add(<X, Y, Z>);
      } else if (_SelectionMode == C_SelectionMode_Plane2D) {
        Result.add(<X, _Start.Y, Z>);
      } else {
        declare Top = _Start.Y;
        declare Bottom = GetFakeGroundHeight(X, Z);
        if (_SelectionMode == C_SelectionMode_Box3D) {
          Top = MathLib::Max(_Start.Y, _End.Y);
          Bottom = MathLib::Min(_Start.Y, _End.Y);
        } else if (_SelectionMode == C_SelectionMode_BoxToGround) {
          if (Bottom != GroundForBox) {
            return [];
          }
          Bottom = GroundForBox;
        }
        if (_SelectionMode == C_SelectionMode_Tower) {
          Top = MathLib::Max(_Start.Y, _End.Y);
        }
        for (Y, Bottom, Top) {
          Result.add(<X, Y, Z>);
        }
      }
    }
  }
  return Result;
}

Void Private_DrawPreview(Int3[] _Coords) {
  CustomSelectionCoords.clear();

  if (G_SelectionVisible) {

  foreach (Coord in _Coords) {

    }
  }
  ShowCustomSelection();
}

Boolean RemoveWater(Int3 _Start, Int3 _End) {
  G_LastRollbackFailed = False;
  log("RemoveWater requested: from (" ^ _Start.X ^ ", " ^ _Start.Y ^ ", " ^ _Start.Z ^ ") to (" ^ _End.X ^ ", " ^ _End.Y ^ ", " ^ _End.Z ^ ").");
  if (TerrainBlockModels.count == 0) {
    log("RemoveWater failed: no terrain block models are available.");
    return False;
  }
  declare Terrain = TerrainBlockModels[0];
  declare MinX = MathLib::Min(_Start.X, _End.X);
  declare MaxX = MathLib::Max(_Start.X, _End.X);
  declare MinZ = MathLib::Min(_Start.Z, _End.Z);
  declare MaxZ = MathLib::Max(_Start.Z, _End.Z);
  declare Int3[] OriginalGroundCoords;
  for (X, MinX, MaxX) {
    for (Z, MinZ, MaxZ) {
      OriginalGroundCoords.add(<X, GetRealGroundHeight(X, Z), Z>);
    }
  }
  if (!PlaceTerrainBlocks(Terrain, _Start, _End)) {
    log("RemoveWater failed: could not place " ^ Terrain.Name ^ " from " ^ _Start ^ " to " ^ _End ^ ".");
    return False;
  }
  declare Succeeded = True;
  declare metadata Int3[] Atlas_RemovedWater as StoredWater for Map;
  foreach (OriginalCoord in OriginalGroundCoords) {
    declare Coord = <OriginalCoord.X, GetRealGroundHeight(OriginalCoord.X, OriginalCoord.Z), OriginalCoord.Z>;
    declare Old = GetBlock(Coord);
    if (Old == Null) {
      log("RemoveWater failed: no block at (" ^ Coord.X ^ ", " ^ Coord.Y ^ ", " ^ Coord.Z ^ ").");
      Succeeded = False;
      continue;
    }
    if (!G_RemoveWaterMapping.existskey(Old.BlockModel.Name)) {
      log("RemoveWater failed: block '" ^ Old.BlockModel.Name ^ "' at (" ^ Coord.X ^ ", " ^ Coord.Y ^ ", " ^ Coord.Z ^ ") has no void mapping.");
      Succeeded = False;
      continue;
    }
    declare VoidName = G_RemoveWaterMapping[Old.BlockModel.Name];
    declare VoidBlock = GetBlockModelFromName(VoidName);
    if (VoidBlock == Null) {
      log("RemoveWater failed: mapped void block '" ^ VoidName ^ "' was not found.");
      Succeeded = False;
      continue;
    }
    if (!PlaceBlock(VoidBlock, Coord, CMapEditorPlugin::CardinalDirections::North)) {
      log("RemoveWater failed: could not place void block '" ^ VoidName ^ "' at (" ^ Coord.X ^ ", " ^ Coord.Y ^ ", " ^ Coord.Z ^ ").");
      Succeeded = False;
      continue;
    }
    if (!StoredWater/* not Null here */.exists(OriginalCoord)) {
      StoredWater.add(OriginalCoord);
    }
  }
  SetCurrentSelection(Private_GetWaterSelectionCoords(StoredWater/* not Null here */), True);
  CustomSelectionRGB = <0.55, 0.30, 0.10>;
  G_GroundItemHeightsValid = False;
  return Succeeded;
}

Boolean RestoreWater(Int3[] _Coords) {
  G_LastRollbackFailed = False;
  if (_Coords.count == 0) {
    return False;
  }
  declare Water = GetBlockModelFromName(G_WaterVoidName);
  if (Water == Null) {
    log("RestoreWater failed: water block '" ^ G_WaterVoidName ^ "' was not found.");
    return False;
  }
  declare metadata Int3[] Atlas_RemovedWater as StoredWater for Map;
  declare Succeeded = True;
  declare Boolean[Int3] AvailableVoidCoords;
  declare Int3[] SelectionCoords;
  declare Boolean[Int3] SelectedCoords;
  declare Boolean[Int3] SelectedColumns;
  foreach (Coord in _Coords) {
    if (!SelectedCoords.existskey(Coord)) {
      SelectionCoords.add(Coord);
      SelectedCoords[Coord] = True;
      SelectedColumns[<Coord.X, 0, Coord.Z>] = True;
    }
  }
  foreach (Block in Blocks) {
    if (Block == Null) {
      continue;
    }
    if (G_RestoreWaterVoidNames.exists(Block.BlockModel.Name)) {
      AvailableVoidCoords[Block.Coord] = True;
    }
  }
  foreach (GroundCoord in SelectionCoords) {
    declare VoidCoord = GroundCoord;
    if (!AvailableVoidCoords.existskey(VoidCoord)) {
      VoidCoord = <GroundCoord.X, CollectionGroundY + 1, GroundCoord.Z>;
    }
    if (!AvailableVoidCoords.existskey(VoidCoord)) {
      continue;
    }
    if (!RemoveBlock(VoidCoord)) {
      log("RestoreWater failed: could not remove configured void block at " ^ VoidCoord ^ ".");
      Succeeded = False;
    }
  }
  declare Int3[] TerrainReadyCoords;
  foreach (GroundCoord in SelectionCoords) {
    if (!RemoveTerrainBlocks(GroundCoord, GroundCoord)) {
      log("RestoreWater failed: could not remove terrain at " ^ GroundCoord ^ ".");
      Succeeded = False;
      continue;
    }
    TerrainReadyCoords.add(GroundCoord);
  }
  declare Boolean[Int3] RestoredColumns;
  foreach (GroundCoord in TerrainReadyCoords) {
    if (!PlaceBlock(Water, GroundCoord, CMapEditorPlugin::CardinalDirections::North)) {
      log("RestoreWater failed: could not place water at " ^ GroundCoord ^ ".");
      Succeeded = False;
      continue;
    }
    RestoredColumns[<GroundCoord.X, 0, GroundCoord.Z>] = True;
  }
  if (RestoredColumns.count > 0) {
    declare Int3[] RemainingWater;
    foreach (StoredCoord in StoredWater/* not Null here */) {
      if (!RestoredColumns.existskey(<StoredCoord.X, 0, StoredCoord.Z>)) {
        RemainingWater.add(StoredCoord);
      }
    }
    StoredWater/* not Null here */.clear();
    foreach (StoredCoord in RemainingWater) {
      StoredWater.add(StoredCoord);
    }
  }
  if (G_RemoveWaterMapping.existskey("Beach")) {
    declare BeachVoidName = G_RemoveWaterMapping["Beach"];
    declare BeachVoid = GetBlockModelFromName(BeachVoidName);
    if (BeachVoid != Null) {
      declare Int3[] OutsideCoords;
      declare Boolean[Int3] OutsideSeen;
      foreach (Coord in SelectionCoords) {
        declare Neighbors = [<Coord.X - 1, Coord.Y, Coord.Z>, <Coord.X + 1, Coord.Y, Coord.Z>, <Coord.X, Coord.Y, Coord.Z - 1>, <Coord.X, Coord.Y, Coord.Z + 1>, <Coord.X - 1, Coord.Y, Coord.Z - 1>, <Coord.X - 1, Coord.Y, Coord.Z + 1>, <Coord.X + 1, Coord.Y, Coord.Z - 1>, <Coord.X + 1, Coord.Y, Coord.Z + 1>, <Coord.X - 2, Coord.Y, Coord.Z - 2>, <Coord.X - 2, Coord.Y, Coord.Z + 2>, <Coord.X + 2, Coord.Y, Coord.Z - 2>, <Coord.X + 2, Coord.Y, Coord.Z + 2>];
        foreach (Neighbor in Neighbors) {
          if (SelectedColumns.existskey(<Neighbor.X, 0, Neighbor.Z>) || OutsideSeen.existskey(Neighbor)) {
            continue;
          }
          OutsideCoords.add(Neighbor);
          OutsideSeen[Neighbor] = True;
        }
      }
      foreach (Coord in OutsideCoords) {
        PlaceBlock(BeachVoid, Coord, CMapEditorPlugin::CardinalDirections::North);
      }
    }
  }
  SetCurrentSelection(StoredWater/* not Null here */, True);
  CustomSelectionRGB = <0.55, 0.30, 0.10>;
  G_GroundItemHeightsValid = False;
  return Succeeded;
}

Void Update() {
  SyncManuallyRemovedItems();
  if (!G_SelectionInputEnabled) {
    G_PreviousMouseDown = Input.MouseLeftButton;
    return;
  }
  declare Pressed = Input.MouseLeftButton;
  if (G_Mode == C_SelectionMode_None) {
    G_PreviousMouseDown = Pressed;
    return;
  }
  declare CursorCoord = Cursor.Coord;
  if (G_Mode == C_SelectionMode_Ground2D || G_Mode == C_SelectionMode_RemoveWater || G_Mode == C_SelectionMode_RestoreWater) {
    CursorCoord = GetMouseCoordOnGround();
    if (G_Mode == C_SelectionMode_RestoreWater) {
      CursorCoord = <CursorCoord.X, CollectionGroundY, CursorCoord.Z>;
    }
  }
  declare DisplayCursorCoord = CursorCoord;
  if (G_Mode == C_SelectionMode_RemoveWater) {
    DisplayCursorCoord = <CursorCoord.X, CollectionGroundY, CursorCoord.Z>;
  }
  Cursor.Coord = DisplayCursorCoord;
  if (G_Mode != C_SelectionMode_None) {
    PlaceMode = CMapEditorPlugin::PlaceMode::CustomSelection;
  }
  if (G_Mode == C_SelectionMode_Tower) {
    declare TowerEnd = <CursorCoord.X + MathLib::Max(1, G_TowerWidth) - 1, CursorCoord.Y, CursorCoord.Z + MathLib::Max(1, G_TowerDepth) - 1>;
    declare TowerCoords = BuildSelection(CursorCoord, TowerEnd, C_SelectionMode_Tower);
    if (Pressed) {
      if (Private_SelectionChangedSinceLastEvent(TowerCoords)) {
        if (G_EmitSelectionChanged) {
          G_SelectionChanged.add(SelectionChange { Start = CursorCoord, End = TowerEnd, Coords = TowerCoords });
        }
        Private_DrawPreview(TowerCoords);
      }
      if (!G_PreviousMouseDown) {
        G_SelectionConfirmed.add(SelectionChange { Start = CursorCoord, End = TowerEnd, Coords = TowerCoords });
      }
    } else if (G_PreviousMouseDown) {
      ClearSelection();
    }
    G_PreviousMouseDown = Pressed;
    return;
  }
  if (Pressed && !G_PreviousMouseDown) {
    G_DragStart = CursorCoord;
    G_Dragging = True;
    Private_ResetSelectionChangeTracking();
  }
  if (G_Dragging) {
    declare Coords = BuildSelection(G_DragStart, CursorCoord, G_Mode);
    declare Change = SelectionChange { Start = G_DragStart, End = CursorCoord, Coords = Coords };
    if (Pressed) {
      if (Private_SelectionChangedSinceLastEvent(Coords)) {
        if (G_EmitSelectionChanged) {
          G_SelectionChanged.add(Change);
        }
        Private_DrawPreview(Coords);
      }
    } else {
      G_SelectionConfirmed.add(Change);
      if (G_Mode == C_SelectionMode_RemoveWater) {
        RemoveWater(G_DragStart, CursorCoord);
      } else if (G_Mode == C_SelectionMode_RestoreWater) {
        RestoreWater(Coords);
      } else {
        ClearSelection();
      }
      G_Dragging = False;
    }
  } else if (G_SelectionVisible) {
    Private_DrawSelection();
  }
  G_PreviousMouseDown = Pressed;
}

Boolean PlaceItemBlockWithFootprint(Text _MacroblockName, Int3 _Coord, CMapEditorPlugin::CardinalDirections _Direction, Boolean _Ground, Text _Family, Integer _PieceIndex, Integer _Width, Integer _Depth) {
  G_LastRollbackFailed = False;
  if (_Width < 1 || _Depth < 1) {
    return False;
  }
  declare Model = GetMacroblockModelFromFilePath(_MacroblockName);
  if (Model == Null) {
    return False;
  }
  declare RemovalResult = Private_RemoveNoItemBlocks(_Coord, _Width, _Depth, _Ground);
  if (!RemovalResult.Success) {
    Private_RestoreNoItemBlocks(RemovalResult.Removed);
    return False;
  }
  if (!CanPlaceMacroblock_NoDestruction(Model, _Coord, _Direction)) {
    Private_RestoreNoItemBlocks(RemovalResult.Removed);
    return False;
  }
  declare PreviousCount = Items.count;
  if (!PlaceMacroblock_NoDestruction(Model, _Coord, _Direction)) {
    Private_RestoreNoItemBlocks(RemovalResult.Removed);
    return False;
  }
  declare Vec3 Position;
  if (Items.count > PreviousCount) {
    Position = Items[PreviousCount].Position;
  } else {
    Position = GetVec3FromCoord(_Coord);
  }
  declare metadata ItemBlock[] Atlas_ItemBlocks as StoredBlocks for Map;
  declare Added = ItemBlock { MacroblockName = _MacroblockName, MacroblockCoord = _Coord, MacroblockDir = DirectionToIndex(_Direction), ItemPosition = Position, Ground = _Ground, Family = _Family, PieceIndex = _PieceIndex, Width = _Width, Depth = _Depth };
  StoredBlocks/* not Null here */.add(Added);
  Private_UpdateGroundItemHeights([], [Added]);
  Private_SnapshotItems();
  Private_PushItemEdit([], [Added], RemovalResult.Removed);
  return True;
}

Boolean PlaceItemBlock(Text _MacroblockName, Int3 _Coord, CMapEditorPlugin::CardinalDirections _Direction, Boolean _Ground, Text _Family, Integer _PieceIndex) {
  return PlaceItemBlockWithFootprint(_MacroblockName, _Coord, _Direction, _Ground, _Family, _PieceIndex, 1, 1);
}

Boolean Private_InsideItemBlock(Int3 _Coord, ItemBlock _Block) {
  return _Coord.Y == _Block.MacroblockCoord.Y && _Coord.X >= _Block.MacroblockCoord.X && _Coord.X < _Block.MacroblockCoord.X + MathLib::Max(1, _Block.Width) && _Coord.Z >= _Block.MacroblockCoord.Z && _Coord.Z < _Block.MacroblockCoord.Z + MathLib::Max(1, _Block.Depth);
}

Integer RemoveItemBlocks(Int3 _Coord, Text _MacroblockName, Integer _Direction) {
  declare metadata ItemBlock[] Atlas_ItemBlocks as StoredBlocks for Map;
  declare ItemBlock[] Remaining;
  declare ItemBlock[] Removed;
  declare Count = 0;
  foreach (Block in StoredBlocks/* not Null here */) {
    if (Private_InsideItemBlock(_Coord, Block) && (_MacroblockName == "" || Block.MacroblockName == _MacroblockName) && (_Direction < 0 || Block.MacroblockDir == _Direction)) {
      declare Model = GetMacroblockModelFromFilePath(Block.MacroblockName);
      if (Model != Null && RemoveMacroblock(Model, Block.MacroblockCoord, DirectionFromIndex(Block.MacroblockDir))) {
        Count += 1;
        Removed.add(Block);
        continue;
      }
    }
    Remaining.add(Block);
  }
  if (Count > 0) {
    SetItemBlockList(Remaining);
  }
  Private_SnapshotItems();
  if (Count > 0) {
    Private_PushItemEdit(Removed, [], []);
  }
  return Count;
}

Boolean Private_HasSide(Integer _Mask, Integer _Side) {
  return _Mask % (_Side * 2) >= _Side;
}

Integer Private_AddSide(Integer __Input_Mask, Integer _Side) {
  declare Integer _Mask = __Input_Mask;
  if (!Private_HasSide(_Mask, _Side)) {
    _Mask += _Side;
  }
  return _Mask;
}

Integer Private_UnionMask(Integer __Input_First, Integer _Second) {
  declare Integer _First = __Input_First;
  if (Private_HasSide(_Second, 1)) {
    _First = Private_AddSide(_First, 1);
  }
  if (Private_HasSide(_Second, 2)) {
    _First = Private_AddSide(_First, 2);
  }
  if (Private_HasSide(_Second, 4)) {
    _First = Private_AddSide(_First, 4);
  }
  if (Private_HasSide(_Second, 8)) {
    _First = Private_AddSide(_First, 8);
  }
  return _First;
}

Integer Private_RotateMask(Integer __Input_Mask, Integer _Turns) {
  declare Integer _Mask = __Input_Mask;
  for (Turn, 0, _Turns - 1) {
    declare Rotated = 0;
    if (Private_HasSide(_Mask, 1)) {
      Rotated += 2;
    }
    if (Private_HasSide(_Mask, 2)) {
      Rotated += 4;
    }
    if (Private_HasSide(_Mask, 4)) {
      Rotated += 8;
    }
    if (Private_HasSide(_Mask, 8)) {
      Rotated += 1;
    }
    _Mask = Rotated;
  }
  return _Mask;
}

Integer RoadPieceForMask(Integer _Mask) {
  if (_Mask == 0) {
    return 0;
  }
  if (_Mask == 15) {
    return 5;
  }
  declare Count = 0;
  if (Private_HasSide(_Mask, 1)) {
    Count += 1;
  }
  if (Private_HasSide(_Mask, 2)) {
    Count += 1;
  }
  if (Private_HasSide(_Mask, 4)) {
    Count += 1;
  }
  if (Private_HasSide(_Mask, 8)) {
    Count += 1;
  }
  if (Count == 1) {
    return 1;
  }
  if (Count == 3) {
    return 4;
  }
  if (_Mask == 5 || _Mask == 10) {
    return 3;
  }
  return 2;
}

CMapEditorPlugin::CardinalDirections RoadDirectionForMask(Integer _Mask) {
  declare Piece = RoadPieceForMask(_Mask);
  declare BaseMask = 0;
  if (Piece == 1) {
    BaseMask = 1;
  }
  if (Piece == 2) {
    BaseMask = 3;
  }
  if (Piece == 3) {
    BaseMask = 5;
  }
  if (Piece == 4) {
    BaseMask = 11;
  }
  if (Piece == 5) {
    BaseMask = 15;
  }
  for (Turn, 0, 4 - 1) {
    if (Private_RotateMask(BaseMask, Turn) == _Mask) {
      return DirectionFromIndex(Turn);
    }
  }
  return CMapEditorPlugin::CardinalDirections::North;
}

    Mask = 5;

  if (_Piece == 4) {

SelectedItemBlockVariant Private_VariantFor(Text _Family, Integer _Piece, Boolean _Ground, Int3 _Coord) {
  if (!G_ItemBlockGroups.existskey(_Family)) {
    return SelectedItemBlockVariant { MacroblockName = "" };
  }
  declare Layers = G_ItemBlockGroups[_Family];
  declare Layer = 0;
  if (_Ground) {
    Layer = 1;
  }
  if (Layer >= Layers.count || _Piece < 0 || _Piece >= Layers[Layer].count) {
    return SelectedItemBlockVariant { MacroblockName = "" };
  }
  declare Variant = Layers[Layer][_Piece];
  if (Variant.Subvariants.count == 0) {
    return SelectedItemBlockVariant { MacroblockName = "" };
  }
  declare Seed = (_Coord.X * 31 + _Coord.Y * 17 + _Coord.Z * 13) % Variant.Subvariants.count;
  if (Seed < 0) {
    Seed += Variant.Subvariants.count;
  }
  return SelectedItemBlockVariant { MacroblockName = Variant.Subvariants[Seed].MacroblockName, DirectionOffset = Variant.DirectionOffset };
}

Integer Private_DirectionOffsetFor(ItemBlock _Block) {
  if (!G_ItemBlockGroups.existskey(_Block.Family) || _Block.PieceIndex < 0) {
    return 0;
  }
  declare Layers = G_ItemBlockGroups[_Block.Family];
  declare Layer = 0;
  if (_Block.Ground) {
    Layer = 1;
  }
  if (Layer >= Layers.count || _Block.PieceIndex >= Layers[Layer].count) {
    return 0;
  }
  declare Variant = Layers[Layer][_Block.PieceIndex];
  foreach (Subvariant in Variant.Subvariants) {
    if (Subvariant.MacroblockName == _Block.MacroblockName) {
      return Variant.DirectionOffset;
    }
  }
  return 0;
}

CMapEditorPlugin::CardinalDirections Private_OffsetDirection(CMapEditorPlugin::CardinalDirections _Direction, Integer _Offset) {
  declare Index = (DirectionToIndex(_Direction) + _Offset) % 4;
  if (Index < 0) {
    Index += 4;
  }
  return DirectionFromIndex(Index);
}

CMapEditorPlugin::CardinalDirections Private_LogicalDirectionFor(ItemBlock _Block) {

}

Integer Private_SideBetween(Int3 _First, Int3 _Second) {
  if (_First.Y != _Second.Y) {
    return 0;
  }
  if (_First.X == _Second.X && _First.Z == _Second.Z + 1) {
    return 1;
  }
  if (_First.X + 1 == _Second.X && _First.Z == _Second.Z) {
    return 2;
  }
  if (_First.X == _Second.X && _First.Z + 1 == _Second.Z) {
    return 4;
  }
  if (_First.X == _Second.X + 1 && _First.Z == _Second.Z) {
    return 8;
  }
  return 0;
}

  if (_Side == 1) {
    return 4;
  }
  if (_Side == 2) {
    return 8;
  }
  if (_Side == 4) {
    return 1;
  }
  if (_Side == 8) {
    return 2;
  }
  return 0;
}

Integer Private_IndexOfCoord(Int3[] _Coords, Int3 _Coord) {
  for (I, 0, _Coords.count - 1) {
    if (Private_SameCoord(_Coords[I], _Coord)) {
      return I;
    }
  }
  return -1;
}

Placement[] ResolveRoadPath(Int3[] _Path, Text _Family) {
  declare Placement[] Result;
  if (_Path.count == 0) {
    return Result;
  }
  declare Int3[] Coords;
  declare Integer[] Masks;
  foreach (Coord in _Path) {
    if (!Private_WithinMap(Coord) || Coord.Y != GetFakeGroundHeight(Coord.X, Coord.Z)) {
      return Result;
    }
    if (Private_IndexOfCoord(Coords, Coord) < 0) {
      Coords.add(Coord);
      Masks.add(0);
    }
  }
  for (I, 1, _Path.count - 1) {
    declare Side = Private_SideBetween(_Path[I - 1], _Path[I]);
    if (Side == 0) {
      return Result;
    }
    declare PreviousIndex = Private_IndexOfCoord(Coords, _Path[I - 1]);
    declare CurrentIndex = Private_IndexOfCoord(Coords, _Path[I]);
    Masks[PreviousIndex] = Private_AddSide(Masks[PreviousIndex], Side);
    Masks[CurrentIndex] = Private_AddSide(Masks[CurrentIndex], Private_OppositeSide(Side));
  }
  declare SelectedCount = Coords.count;
  for (I, 0, SelectedCount - 1) {
    foreach (Old in GetItemBlockList()) {
      if (Old.Family != _Family || Old.MacroblockCoord.Y != Coords[I].Y) {
        continue;
      }
      declare Side = Private_SideBetween(Coords[I], Old.MacroblockCoord);
      if (Side == 0) {
        continue;
      }
      declare NeighborIndex = Private_IndexOfCoord(Coords, Old.MacroblockCoord);
      if (NeighborIndex < 0) {
        NeighborIndex = Coords.count;
        Coords.add(Old.MacroblockCoord);
        Masks.add(0);
      }
      Masks[I] = Private_AddSide(Masks[I], Side);
      Masks[NeighborIndex] = Private_AddSide(Masks[NeighborIndex], Private_OppositeSide(Side));
    }
  }
  for (I, 0, Coords.count - 1) {
    foreach (Old in GetItemBlockList()) {
      if (!Private_SameCoord(Old.MacroblockCoord, Coords[I])) {
        continue;
      }
      if (Old.Family != _Family) {
        return [];
      }
      declare OldMask = Private_MaskForRoadPiece(Old.PieceIndex, Private_LogicalDirectionFor(Old));
      if (Private_HasSide(OldMask, 1)) {
        Masks[I] = Private_AddSide(Masks[I], 1);
      }
      if (Private_HasSide(OldMask, 2)) {
        Masks[I] = Private_AddSide(Masks[I], 2);
      }
      if (Private_HasSide(OldMask, 4)) {
        Masks[I] = Private_AddSide(Masks[I], 4);
      }
      if (Private_HasSide(OldMask, 8)) {
        Masks[I] = Private_AddSide(Masks[I], 8);
      }
    }
    declare Ground = Coords[I].Y == GetFakeGroundHeight(Coords[I].X, Coords[I].Z);
    declare Piece = RoadPieceForMask(Masks[I]);
    declare Variant = Private_VariantFor(_Family, Piece, Ground, Coords[I]);
    if (Variant.MacroblockName == "") {
      return [];
    }
    Result.add(Placement { Coord = Coords[I], MacroblockName = Variant.MacroblockName, Direction = Private_OffsetDirection(RoadDirectionForMask(Masks[I]), Variant.DirectionOffset), Ground = Ground, Family = _Family, PieceIndex = Piece });
  }
  return Result;
}

Boolean PlaceRoadPath(Int3[] _Path, Text _Family) {
  return ExecutePlacementPlan(ResolveRoadPath(_Path, _Family));
}

Boolean Private_IsFreeformBase(Integer _Piece) {
  return _Piece >= 0 && (_Piece <= 4 || _Piece == 14);
}

Integer Private_FreeformSideMask(Integer _Piece, CMapEditorPlugin::CardinalDirections _Direction) {
  declare Mask = 0;
  if (Private_IsFreeformBase(_Piece)) {
    Mask = 15;
  } else if (_Piece >= 5 && _Piece <= 8) {
    Mask = 11;
  } else {
    if (_Piece == 9 || _Piece == 10) {
      Mask = 3;
    } else if (_Piece == 11) {
      Mask = 5;
    } else {
      if (_Piece == 12) {
        Mask = 1;
      }
    }
  }
  return Private_RotateMask(Mask, DirectionToIndex(_Direction));
}

Integer Private_FreeformDiagonalMask(Integer _Piece, CMapEditorPlugin::CardinalDirections _Direction) {
  declare Mask = 0;
  if (_Piece == 0) {
    Mask = 14;
  } else if (_Piece == 1) {
    Mask = 12;
  } else {
    if (_Piece == 2) {
      Mask = 10;
    } else if (_Piece == 3) {
      Mask = 8;
    } else {
      if (_Piece == 5) {
        Mask = 3;
      } else if (_Piece == 6) {
        Mask = 2;
      } else {
        if (_Piece == 7) {
          Mask = 1;
        } else if (_Piece == 9) {
          Mask = 2;
        } else {
          if (_Piece == 14) {
            Mask = 15;
          }
        }
      }
    }
  }
  return Private_RotateMask(Mask, DirectionToIndex(_Direction));
}

Integer Private_FreeformSideCount(Integer _Mask) {
  declare Count = 0;
  if (Private_HasSide(_Mask, 1)) {
    Count += 1;
  }
  if (Private_HasSide(_Mask, 2)) {
    Count += 1;
  }
  if (Private_HasSide(_Mask, 4)) {
    Count += 1;
  }
  if (Private_HasSide(_Mask, 8)) {
    Count += 1;
  }
  return Count;
}

CMapEditorPlugin::CardinalDirections Private_DirectionForDiagonalMask(Integer _Mask, Integer _NorthMask) {
  for (Turn, 0, 4 - 1) {
    if (Private_RotateMask(_NorthMask, Turn) == _Mask) {
      return DirectionFromIndex(Turn);
    }
  }
  return CMapEditorPlugin::CardinalDirections::North;
}

FreeformCandidate Private_ResolveFreeformMasks(Integer _Mask, Integer _Diagonals, Boolean _HasFiller) {
  declare MissingDiagonals = 15 - _Diagonals;
  declare Sides = Private_FreeformSideCount(_Mask);
  declare Direction = RoadDirectionForMask(_Mask);
  declare Piece = 13;
  if (Sides == 1) {
    Piece = 12;
  } else if (Sides == 2) {
    if (_Mask == 5 || _Mask == 10) {
      Piece = 11;
    } else {
      declare Normalized = Private_RotateMask(MissingDiagonals, (4 - DirectionToIndex(Direction)) % 4);
      if (Private_HasSide(Normalized, 2)) {
        Piece = 10;
      } else {
        Piece = 9;
      }
    }
  } else {
    if (Sides == 3) {
      declare Normalized = Private_RotateMask(MissingDiagonals, (4 - DirectionToIndex(Direction)) % 4);
      declare MissingLeft = Private_HasSide(Normalized, 1);
      declare MissingRight = Private_HasSide(Normalized, 2);
      Piece = 5;
      if (MissingLeft && MissingRight) {
        Piece = 8;
      } else if (MissingLeft) {
        Piece = 6;
      } else {
        if (MissingRight) {
          Piece = 7;
        }
      }
    } else if (Sides == 4) {
      if (MissingDiagonals == 0) {
        if (_HasFiller) {
          Piece = 14;
        } else {
          Piece = 4;
        }
      } else if (MissingDiagonals == 15) {
        Piece = 4;
      } else {
        declare MissingCount = 0;
        if (Private_HasSide(MissingDiagonals, 1)) {
          MissingCount += 1;
        }
        if (Private_HasSide(MissingDiagonals, 2)) {
          MissingCount += 1;
        }
        if (Private_HasSide(MissingDiagonals, 4)) {
          MissingCount += 1;
        }
        if (Private_HasSide(MissingDiagonals, 8)) {
          MissingCount += 1;
        }
        if (MissingCount == 1) {
          Piece = 0;
          Direction = Private_DirectionForDiagonalMask(MissingDiagonals, 1);
        } else if (MissingCount == 2 && (MissingDiagonals == 5 || MissingDiagonals == 10)) {
          Piece = 2;
          Direction = Private_DirectionForDiagonalMask(MissingDiagonals, 5);
        } else {
          if (MissingCount == 2) {
            Piece = 1;
            Direction = Private_DirectionForDiagonalMask(MissingDiagonals, 3);
          } else {
            Piece = 3;
            Direction = Private_DirectionForDiagonalMask(MissingDiagonals, 7);
          }
        }
      }
    }
  }
  return FreeformCandidate { Piece = Piece, Direction = Direction };
}

FreeformCandidate ResolveFreeformOverlap(Integer _OldPiece, CMapEditorPlugin::CardinalDirections _OldDirection, Integer _NewPiece, CMapEditorPlugin::CardinalDirections _NewDirection) {
  declare Sides = Private_UnionMask(Private_FreeformSideMask(_OldPiece, _OldDirection), Private_FreeformSideMask(_NewPiece, _NewDirection));
  declare Diagonals = Private_UnionMask(Private_FreeformDiagonalMask(_OldPiece, _OldDirection), Private_FreeformDiagonalMask(_NewPiece, _NewDirection));
  return Private_ResolveFreeformMasks(Sides, Diagonals, _OldPiece == 14 || _NewPiece == 14);
}

Integer Private_FreeformOccupiedSides(Boolean[Int3] _Occupied, Int3 _Coord) {
  declare Mask = 0;
  if (_Occupied.existskey(<_Coord.X, _Coord.Y, _Coord.Z - 1>)) {
    Mask += 1;
  }
  if (_Occupied.existskey(<_Coord.X + 1, _Coord.Y, _Coord.Z>)) {
    Mask += 2;
  }
  if (_Occupied.existskey(<_Coord.X, _Coord.Y, _Coord.Z + 1>)) {
    Mask += 4;
  }
  if (_Occupied.existskey(<_Coord.X - 1, _Coord.Y, _Coord.Z>)) {
    Mask += 8;
  }
  return Mask;
}

Integer Private_FreeformOccupiedDiagonals(Boolean[Int3] _Occupied, Int3 _Coord) {
  declare Mask = 0;
  if (_Occupied.existskey(<_Coord.X - 1, _Coord.Y, _Coord.Z - 1>)) {
    Mask += 1;
  }
  if (_Occupied.existskey(<_Coord.X + 1, _Coord.Y, _Coord.Z - 1>)) {
    Mask += 2;
  }
  if (_Occupied.existskey(<_Coord.X + 1, _Coord.Y, _Coord.Z + 1>)) {
    Mask += 4;
  }
  if (_Occupied.existskey(<_Coord.X - 1, _Coord.Y, _Coord.Z + 1>)) {
    Mask += 8;
  }
  return Mask;
}

FreeformCandidate ResolveFreeformCell(Boolean[Int3] _Occupied, Boolean[Int3] _Selected, Int3 _Coord, Integer _OldPiece, CMapEditorPlugin::CardinalDirections _OldDirection, Boolean _HasFiller) {
  declare Sides = Private_FreeformOccupiedSides(_Occupied, _Coord);
  declare Diagonals = Private_FreeformOccupiedDiagonals(_Occupied, _Coord);
  if (_OldPiece >= 0 && (_Selected.existskey(_Coord) || Private_IsFreeformBase(_OldPiece))) {
    Sides = Private_UnionMask(Sides, Private_FreeformSideMask(_OldPiece, _OldDirection));
    Diagonals = Private_UnionMask(Diagonals, Private_FreeformDiagonalMask(_OldPiece, _OldDirection));
  }
  if (_OldPiece >= 0 && _Selected.existskey(_Coord) && Private_FreeformSideCount(Private_FreeformOccupiedSides(_Selected, _Coord)) >= 3 && Private_FreeformSideCount(Sides) == 3) {
    declare EdgeDirection = RoadDirectionForMask(Sides);
    Diagonals = Private_UnionMask(Diagonals, Private_RotateMask(3, DirectionToIndex(EdgeDirection)));
  }
  return Private_ResolveFreeformMasks(Sides, Diagonals, _HasFiller);
}

Placement[] ResolveFreeform1x1(Int3[] _Selection, Text _Family) {
  declare Placement[] Result;
  if (_Selection.count == 0) {
    return Result;
  }
  declare MinX = _Selection[0].X;
  declare MaxX = MinX;
  declare MinZ = _Selection[0].Z;
  declare MaxZ = MinZ;
  declare Boolean[Int3] SelectedLookup;
  declare Int3[Int3] SelectedByCell;
  foreach (Coord in _Selection) {
    if (!Private_WithinMap(Coord) || Coord.Y != GetFakeGroundHeight(Coord.X, Coord.Z)) {
      return [];
    }
    declare Cell = <Coord.X, 0, Coord.Z>;
    if (SelectedByCell.existskey(Cell)) {
      return Result;
    }
    SelectedByCell[Cell] = Coord;
    SelectedLookup[Coord] = True;
    MinX = MathLib::Min(MinX, Coord.X);
    MaxX = MathLib::Max(MaxX, Coord.X);
    MinZ = MathLib::Min(MinZ, Coord.Z);
    MaxZ = MathLib::Max(MaxZ, Coord.Z);
  }
  if (_Selection.count != (MaxX - MinX + 1) * (MaxZ - MinZ + 1)) {
    return Result;
  }
  declare Int3[] Selected;
  for (X, MinX, MaxX) {
    for (Z, MinZ, MaxZ) {
      declare Cell = <X, 0, Z>;
      if (!SelectedByCell.existskey(Cell)) {
        return [];
      }
      Selected.add(SelectedByCell[Cell]);
    }
  }
  declare Existing = GetItemBlockList();
  declare Boolean[Int3] Occupied;
  declare Int3[] Affected;
  declare Boolean[Int3] AffectedLookup;
  declare ItemBlock[Int3] OldByCoord;
  declare Boolean[Int3] DuplicateCoords;
  foreach (Coord in Selected) {
    Affected.add(Coord);
    AffectedLookup[Coord] = True;
  }
  foreach (Old in Existing) {
    declare OldCoord = Old.MacroblockCoord;
    if (OldCoord.X >= MinX - 1 && OldCoord.X <= MaxX + 1 && OldCoord.Z >= MinZ - 1 && OldCoord.Z <= MaxZ + 1) {
      if (OldByCoord.existskey(OldCoord)) {
        DuplicateCoords[OldCoord] = True;
      } else {
        OldByCoord[OldCoord] = Old;
      }
    }
    if (OldCoord.X <= MaxX && OldCoord.X + MathLib::Max(1, Old.Width) > MinX && OldCoord.Z <= MaxZ && OldCoord.Z + MathLib::Max(1, Old.Depth) > MinZ) {
      foreach (Coord in Selected) {
        if (Private_InsideItemBlock(Coord, Old) && (!Old.Ground || Old.Width != 1 || Old.Depth != 1)) {
          return [];
        }
      }
    }
    if (G_FreeformPlacementMode == C_FreeformPlacementMode_SelectionOnly) {
      continue;
    }
    if (Old.Family != _Family || OldCoord.X < MinX - 2 || OldCoord.X > MaxX + 2 || OldCoord.Z < MinZ - 2 || OldCoord.Z > MaxZ + 2) {
      continue;
    }
    declare Nearby = False;
    declare TouchesSelection = False;
    for (Dx, -2, 2) {
      for (Dz, -2, 2) {
        if (!SelectedLookup.existskey(<OldCoord.X + Dx, OldCoord.Y, OldCoord.Z + Dz>)) {
          continue;
        }
        Nearby = True;
        if (MathLib::Abs(Dx) <= 1 && MathLib::Abs(Dz) <= 1) {
          TouchesSelection = True;
        }
      }
    }
    if (Nearby && (!Old.Ground || Old.Width != 1 || Old.Depth != 1)) {
      return [];
    }
    if (Old.Ground && Old.Width == 1 && Old.Depth == 1) {
      Occupied[OldCoord] = True;
      if (TouchesSelection && !AffectedLookup.existskey(OldCoord)) {
        Affected.add(OldCoord);
        AffectedLookup[OldCoord] = True;
      }
    }
  }
  foreach (Coord in Selected) {
    Occupied[Coord] = True;
  }
  declare HasFiller = Private_VariantFor(_Family, 14, True, Selected[0]).MacroblockName != "";
  foreach (Coord in Affected) {
    declare HasOld = OldByCoord.existskey(Coord);
    declare ItemBlock OldBlock;
    if (HasOld) {
      OldBlock = OldByCoord[Coord];
    } else {
      OldBlock = ItemBlock { MacroblockName = "" };
    }
    if (HasOld && (OldBlock.Family != _Family || DuplicateCoords.existskey(Coord))) {
      return [];
    }
    declare OldLogicalDirection = CMapEditorPlugin::CardinalDirections::North;
    if (HasOld) {
      OldLogicalDirection = Private_LogicalFreeformDirectionFor(OldBlock);
    }
    declare OldPiece = -1;
    if (HasOld) {
      OldPiece = OldBlock.PieceIndex;
    }
    declare Desired = ResolveFreeformCell(Occupied, SelectedLookup, Coord, OldPiece, OldLogicalDirection, HasFiller);
    if (HasOld && OldBlock.PieceIndex == Desired.Piece && OldLogicalDirection == Desired.Direction) {
      Result.add(Placement { Coord = Coord, MacroblockName = OldBlock.MacroblockName, Direction = DirectionFromIndex(OldBlock.MacroblockDir), Ground = True, Family = _Family, PieceIndex = OldBlock.PieceIndex, Width = 1, Depth = 1 });
      continue;
    }
    declare Variant = Private_VariantFor(_Family, Desired.Piece, True, Coord);
    if (Variant.MacroblockName == "") {
      return [];
    }
    Result.add(Placement { Coord = Coord, MacroblockName = Variant.MacroblockName, Direction = Private_OffsetDirection(Desired.Direction, Variant.DirectionOffset + Private_FreeformModelDirectionOffset(Desired.Piece)), Ground = True, Family = _Family, PieceIndex = Desired.Piece, Width = 1, Depth = 1 });
  }
  return Result;
}

Boolean PlaceFreeform1x1(Int3[] _Selection, Text _Family) {
  return ExecutePlacementPlan(ResolveFreeform1x1(_Selection, _Family));
}

Placement[] ResolveCube2x2(Int3[] _Selection, Text _Family) {
  declare Placement[] Result;
  if (_Selection.count == 0) {
    return Result;
  }
  declare MinX = _Selection[0].X;
  declare MaxX = MinX;
  declare MinY = _Selection[0].Y;
  declare MaxY = MinY;
  declare MinZ = _Selection[0].Z;
  declare MaxZ = MinZ;
  foreach (Coord in _Selection) {
    MinX = MathLib::Min(MinX, Coord.X);
    MaxX = MathLib::Max(MaxX, Coord.X);
    MinY = MathLib::Min(MinY, Coord.Y);
    MaxY = MathLib::Max(MaxY, Coord.Y);
    MinZ = MathLib::Min(MinZ, Coord.Z);
    MaxZ = MathLib::Max(MaxZ, Coord.Z);
  }
  declare SizeX = MaxX - MinX + 1;
  declare SizeZ = MaxZ - MinZ + 1;
  if (SizeX < 2 || SizeZ < 2 || SizeX != SizeZ || SizeX % 2 != 0 || _Selection.count != SizeX * SizeZ * (MaxY - MinY + 1)) {
    return Result;
  }
  foreach (Coord in _Selection) {
    if (!Private_WithinMap(Coord)) {
      return [];
    }
  }
  for (X, MinX, MaxX) {
    for (Z, MinZ, MaxZ) {
      declare Ground = GetFakeGroundHeight(X, Z);
      if (Ground != MinY) {
        return [];
      }
      for (Y, MinY, MaxY) {
        if (Private_IndexOfCoord(_Selection, <X, Y, Z>) < 0) {
          return [];
        }
      }
    }
  }
  declare X = MinX;
  while (X <= MaxX) {
    declare Z = MinZ;
    while (Z <= MaxZ) {
      declare Mask = 0;
      if (Z > MinZ) {
        Mask += 1;
      }
      if (X < MaxX - 1) {
        Mask += 2;
      }
      if (Z < MaxZ - 1) {
        Mask += 4;
      }
      if (X > MinX) {
        Mask += 8;
      }
      if (!G_CubePieceMapping.existskey(_Family) || !G_CubePieceMapping[_Family].existskey(Mask)) {
        return [];
      }
      declare Piece = G_CubePieceMapping[_Family][Mask];
      for (Y, MinY, MaxY) {
        declare Layer = _Family ^ "#middle";
        if (Y == MinY) {
          Layer = _Family ^ "#bottom";
        }
        if (Y == MaxY) {
          Layer = _Family ^ "#top";
        }
        declare Anchor = <X, Y, Z>;
        declare Variant = Private_VariantFor(Layer, Piece, Y == MinY, Anchor);
        if (Variant.MacroblockName == "") {
          return [];
        }
        Result.add(Placement { Coord = Anchor, MacroblockName = Variant.MacroblockName, Direction = Private_OffsetDirection(CMapEditorPlugin::CardinalDirections::North, Variant.DirectionOffset), Ground = Y == MinY, Family = _Family, PieceIndex = Piece, Width = 2, Depth = 2 });
      }
      Z += 2;
    }
    X += 2;
  }
  return Result;
}

Boolean PlaceCube2x2(Int3[] _Selection, Text _Family) {
  return ExecutePlacementPlan(ResolveCube2x2(_Selection, _Family));
}

Placement[] ResolveTower(Int3[] _Selection, Text _Family, Integer _Width, Integer _Depth) {
  declare Placement[] Result;
  if (_Selection.count == 0 || _Width < 1 || _Depth < 1) {
    return Result;
  }
  declare MinX = _Selection[0].X;
  declare MaxX = MinX;
  declare MinY = _Selection[0].Y;
  declare MaxY = MinY;
  declare MinZ = _Selection[0].Z;
  declare MaxZ = MinZ;
  foreach (Coord in _Selection) {
    MinX = MathLib::Min(MinX, Coord.X);
    MaxX = MathLib::Max(MaxX, Coord.X);
    MinY = MathLib::Min(MinY, Coord.Y);
    MaxY = MathLib::Max(MaxY, Coord.Y);
    MinZ = MathLib::Min(MinZ, Coord.Z);
    MaxZ = MathLib::Max(MaxZ, Coord.Z);
  }
  if ((MaxX - MinX + 1) % _Width != 0 || (MaxZ - MinZ + 1) % _Depth != 0 || _Selection.count != (MaxX - MinX + 1) * (MaxZ - MinZ + 1) * (MaxY - MinY + 1)) {
    return Result;
  }
  for (X, MinX, MaxX) {
    for (Z, MinZ, MaxZ) {
      for (Y, MinY, MaxY) {
        if (Private_IndexOfCoord(_Selection, <X, Y, Z>) < 0 || GetFakeGroundHeight(X, Z) != MinY) {
          return [];
        }
      }
    }
  }
  declare X = MinX;
  while (X <= MaxX) {
    declare Z = MinZ;
    while (Z <= MaxZ) {
      for (Y, MinY, MaxY) {
        declare Anchor = <X, Y, Z>;
        declare Layer = _Family ^ "#middle";
        if (Y == MinY) {
          Layer = _Family ^ "#bottom";
        }
        if (Y == MaxY) {
          Layer = _Family ^ "#top";
        }
        declare Variant = Private_VariantFor(Layer, 0, Y == MinY, Anchor);
        if (Variant.MacroblockName == "") {
          return [];
        }
        Result.add(Placement { Coord = Anchor, MacroblockName = Variant.MacroblockName, Direction = Private_OffsetDirection(CMapEditorPlugin::CardinalDirections::North, Variant.DirectionOffset), Ground = Y == MinY, Family = _Family, PieceIndex = 0, Width = _Width, Depth = _Depth });
      }
      Z += _Depth;
    }
    X += _Width;
  }
  return Result;
}

Boolean PlaceTower(Int3[] _Selection, Text _Family, Integer _Width, Integer _Depth) {
  return ExecutePlacementPlan(ResolveTower(_Selection, _Family, _Width, _Depth));
}

Placement[] ResolveFreeform2x2(Int3[] _Selection, Text _Family, Integer _Height) {
  declare Placement[] Result;
  if (_Selection.count < 4 || _Height < 1) {
    return Result;
  }
  declare MinX = _Selection[0].X;
  declare MinZ = _Selection[0].Z;
  foreach (Coord in _Selection) {
    MinX = MathLib::Min(MinX, Coord.X);
    MinZ = MathLib::Min(MinZ, Coord.Z);
  }
  declare Int3[] Anchors;
  foreach (Coord in _Selection) {
    declare X = MinX + (Coord.X - MinX) / 2 * 2;
    declare Z = MinZ + (Coord.Z - MinZ) / 2 * 2;
    declare Anchor = <X, Coord.Y, Z>;
    if (Private_IndexOfCoord(Anchors, Anchor) < 0) {
      Anchors.add(Anchor);
    }
  }
  declare Int3[] Connected;
  Connected.add(Anchors[0]);
  declare I = 0;
  while (I < Connected.count) {
    declare Anchor = Connected[I];
    declare North = <Anchor.X, Anchor.Y, Anchor.Z - 2>;
    declare East = <Anchor.X + 2, Anchor.Y, Anchor.Z>;
    declare South = <Anchor.X, Anchor.Y, Anchor.Z + 2>;
    declare West = <Anchor.X - 2, Anchor.Y, Anchor.Z>;
    if (Private_IndexOfCoord(Anchors, North) >= 0 && Private_IndexOfCoord(Connected, North) < 0) {
      Connected.add(North);
    }
    if (Private_IndexOfCoord(Anchors, East) >= 0 && Private_IndexOfCoord(Connected, East) < 0) {
      Connected.add(East);
    }
    if (Private_IndexOfCoord(Anchors, South) >= 0 && Private_IndexOfCoord(Connected, South) < 0) {
      Connected.add(South);
    }
    if (Private_IndexOfCoord(Anchors, West) >= 0 && Private_IndexOfCoord(Connected, West) < 0) {
      Connected.add(West);
    }
    I += 1;
  }
  if (Connected.count != Anchors.count) {
    return [];
  }
  foreach (Anchor in Anchors) {
    for (X, Anchor.X, Anchor.X + 2 - 1) {
      for (Z, Anchor.Z, Anchor.Z + 2 - 1) {
        if (!Private_WithinMap(<X, Anchor.Y, Z>) || Private_IndexOfCoord(_Selection, <X, Anchor.Y, Z>) < 0 || GetFakeGroundHeight(X, Z) != Anchor.Y) {
          return [];
        }
      }
    }
    declare Mask = 0;
    if (Private_IndexOfCoord(Anchors, <Anchor.X, Anchor.Y, Anchor.Z - 2>) >= 0) {
      Mask += 1;
    }
    if (Private_IndexOfCoord(Anchors, <Anchor.X + 2, Anchor.Y, Anchor.Z>) >= 0) {
      Mask += 2;
    }
    if (Private_IndexOfCoord(Anchors, <Anchor.X, Anchor.Y, Anchor.Z + 2>) >= 0) {
      Mask += 4;
    }
    if (Private_IndexOfCoord(Anchors, <Anchor.X - 2, Anchor.Y, Anchor.Z>) >= 0) {
      Mask += 8;
    }
    declare Piece = 3;
    if (Mask == 0 || Mask == 15) {
      Piece = 0;
    } else if (Mask == 3 || Mask == 6 || Mask == 9 || Mask == 12) {
      Piece = 2;
    } else {
      if (Mask == 7 || Mask == 11 || Mask == 13 || Mask == 14) {
        Piece = 1;
      }
    }
    declare Direction = RoadDirectionForMask(Mask);
    for (LayerIndex, 0, _Height - 1) {
      declare Y = Anchor.Y + LayerIndex;
      declare Layer = _Family ^ "#middle";
      if (LayerIndex == 0) {
        Layer = _Family ^ "#bottom";
      }
      if (LayerIndex == _Height - 1) {
        Layer = _Family ^ "#top";
      }
      declare PlacedAt = <Anchor.X, Y, Anchor.Z>;
      declare Variant = Private_VariantFor(Layer, Piece, LayerIndex == 0, PlacedAt);
      if (Variant.MacroblockName == "") {
        return [];
      }
      Result.add(Placement { Coord = PlacedAt, MacroblockName = Variant.MacroblockName, Direction = Private_OffsetDirection(Direction, Variant.DirectionOffset), Ground = LayerIndex == 0, Family = _Family, PieceIndex = Piece, Width = 2, Depth = 2 });
    }
  }
  return Result;
}

Boolean PlaceFreeform2x2(Int3[] _Selection, Text _Family, Integer _Height) {
  return ExecutePlacementPlan(ResolveFreeform2x2(_Selection, _Family, _Height));
}
